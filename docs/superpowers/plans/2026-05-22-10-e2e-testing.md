```typescript
  },
});
```

- [ ] **Step 3: Create Electron launch helper**

Create `tests/e2e/helpers.ts`:

```typescript
import { _electron as electron } from "playwright";
import type { ElectronApplication, Page } from "playwright";

export async function launchApp(): Promise<{
  app: ElectronApplication;
  page: Page;
}> {
  const app = await electron.launch({
    args: ["."], // relative to project root
    env: {
      ...process.env,
      VITE_DEV_SERVER_URL: "", // Use built assets, not dev server
    },
  });
  const page = await app.firstWindow();
  return { app, page };
}

export async function closeApp(app: ElectronApplication): Promise<void> {
  await app.close();
}
```

### Task 2: Write E2E test for app launch and home view

**Files:**

- Create: `tests/e2e/home.spec.ts`

- [ ] **Step 1: Write home view E2E test**

Create `tests/e2e/home.spec.ts`:

```typescript
import { test, expect } from "@playwright/test";
import { launchApp, closeApp } from "./helpers";
import type { ElectronApplication, Page } from "playwright";

let app: ElectronApplication;
let page: Page;

test.beforeAll(async () => {
  const result = await launchApp();
  app = result.app;
  page = result.page;
});

test.afterAll(async () => {
  await closeApp(app);
});

test("app launches and shows home view", async () => {
  await expect(page.locator("text=Welcome to UnSqitch")).toBeVisible({
    timeout: 10000,
  });
});

test("shows Open a Project button", async () => {
  await expect(page.locator("text=Open a Project")).toBeVisible();
});

test("shows New Project link", async () => {
  await expect(page.locator("text=create a new project")).toBeVisible();
});

test("sidebar shows UnSqitch branding", async () => {
  await expect(page.locator("text=UnSqitch")).toBeVisible();
});
```
